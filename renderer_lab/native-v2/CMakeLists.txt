cmake_minimum_required(VERSION 3.25)
project(SaeedRendererLabV2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED FILAMENT_ROOT OR NOT DEFINED BGFX_ROOT OR NOT DEFINED BX_ROOT OR NOT DEFINED BIMG_ROOT)
 message(FATAL_ERROR "Missing renderer roots")
endif()
add_subdirectory("${FILAMENT_ROOT}" "${CMAKE_BINARY_DIR}/filament-build" EXCLUDE_FROM_ALL)
add_library(bgfx_shared SHARED IMPORTED GLOBAL)
set_target_properties(bgfx_shared PROPERTIES IMPORTED_IMPLIB_RELEASE "${BGFX_ROOT}/.build/win64_vs2022/bin/bgfx-shared-libRelease.lib" IMPORTED_LOCATION_RELEASE "${BGFX_ROOT}/.build/win64_vs2022/bin/bgfx-shared-libRelease.dll" INTERFACE_INCLUDE_DIRECTORIES "${BGFX_ROOT}/include;${BX_ROOT}/include;${BIMG_ROOT}/include")
add_executable(Saeed-Renderer-Bgfx-D3D11 WIN32 ../native/v2_bgfx_main.cpp)
add_executable(Saeed-Renderer-Bgfx-Vulkan WIN32 ../native/v2_bgfx_main.cpp)
add_executable(Saeed-Renderer-Filament WIN32 ../native/v2_filament_main.cpp)
add_executable(Saeed-3Screen-V2 WIN32 ../native/v2_launcher.cpp)
foreach(x Saeed-Renderer-Bgfx-D3D11 Saeed-Renderer-Bgfx-Vulkan)
 target_link_libraries(${x} PRIVATE bgfx_shared d3d11 dxgi opengl32)
 target_include_directories(${x} PRIVATE "${BGFX_ROOT}/include" "${BX_ROOT}/include" "${BIMG_ROOT}/include")
 target_compile_definitions(${x} PRIVATE UNICODE _UNICODE NOMINMAX)
endforeach()
target_link_libraries(Saeed-Renderer-Filament PRIVATE filament backend utils gltfio)
target_compile_definitions(Saeed-Renderer-Filament PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(Saeed-3Screen-V2 PRIVATE urlmon)
target_compile_definitions(Saeed-3Screen-V2 PRIVATE UNICODE _UNICODE NOMINMAX)
