cmake_minimum_required(VERSION 3.25)
project(SaeedRendererLabNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(cgltf URL https://github.com/jkuhlmann/cgltf/archive/refs/tags/v1.14.tar.gz)
FetchContent_MakeAvailable(cgltf)
if(NOT FILAMENT_ROOT)
  message(FATAL_ERROR "FILAMENT_ROOT is required for the real Filament window")
endif()
add_subdirectory($ENV{BGFX_CMAKE_ROOT} \${CMAKE_BINARY_DIR}/bgfx-build EXCLUDE_FROM_ALL)
add_subdirectory(\${FILAMENT_ROOT} \${CMAKE_BINARY_DIR}/filament-build EXCLUDE_FROM_ALL)
add_executable(saeed_renderer_lab_native WIN32 main.cpp d3d11_window.cpp opengl_window.cpp filament_window.cpp bgfx_window.cpp)
target_include_directories(saeed_renderer_lab_native PRIVATE \${cgltf_SOURCE_DIR})
target_link_libraries(saeed_renderer_lab_native PRIVATE d3d11 d3dcompiler dxgi opengl32 bgfx filament backend utils gltfio)
target_compile_definitions(saeed_renderer_lab_native PRIVATE UNICODE _UNICODE NOMINMAX SAEED_RENDERER_LAB_REAL_BACKENDS=1)
